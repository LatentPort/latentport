cmake_minimum_required(VERSION 3.24)
project(latentport_native LANGUAGES C CXX CUDA)
if(NOT LATENTPORT_LLAMA_SOURCE)
  message(FATAL_ERROR "Set LATENTPORT_LLAMA_SOURCE to a patched pinned llama.cpp checkout")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_CUDA ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
find_package(CUDAToolkit REQUIRED)
add_subdirectory("${LATENTPORT_LLAMA_SOURCE}" llama-build)
add_library(latentport_layout OBJECT layout.cu)
set_target_properties(latentport_layout PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(latentport_layout PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_sources(llama PRIVATE latentport.cpp $<TARGET_OBJECTS:latentport_layout>)
target_include_directories(llama PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(llama PRIVATE CUDA::cudart)
enable_testing()
add_executable(latentport-layout-test test_layout.cpp)
add_test(NAME latentport-layout COMMAND latentport-layout-test)
